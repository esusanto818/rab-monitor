--
-- Name: view_employee_depart_status; Type: VIEW; Schema: public; Owner: postgres
--

CREATE VIEW public.view_employee_depart_status AS
 SELECT e.id,
    e.employee_name,
    d.id AS employee_department_id,
    l.id AS location_id,
    l.name AS location_name,
        CASE
            WHEN (lower(btrim((l.name)::text)) ~~ '%office%'::text) THEN md.init
            ELSE l.init
        END AS location_init,
    md.id AS department_id,
    md.name AS department_name,
    md.init AS department_init,
    sc.id AS employee_section_id,
    ds.id AS section_id,
    ds.section_name,
    s.id AS employee_status_id,
    ms.id AS status_id,
    ms.name AS status_name
   FROM (((((((public.m_employee e
     LEFT JOIN ( SELECT DISTINCT ON (employee_department.employee_id) employee_department.id,
            employee_department.employee_id,
            employee_department.department_id,
            employee_department.from_date
           FROM public.employee_department
          WHERE (employee_department.to_date IS NULL)
          ORDER BY employee_department.employee_id, employee_department.from_date DESC, employee_department.id DESC) d ON ((d.employee_id = e.id)))
     LEFT JOIN public.m_department md ON ((md.id = d.department_id)))
     LEFT JOIN public.m_location l ON ((l.id = md.location_id)))
     LEFT JOIN ( SELECT DISTINCT ON (employee_section.employee_id) employee_section.id,
            employee_section.employee_id,
            employee_section.section_id,
            employee_section.from_date
           FROM public.employee_section
          WHERE (employee_section.to_date IS NULL)
          ORDER BY employee_section.employee_id, employee_section.from_date DESC, employee_section.id DESC) sc ON ((sc.employee_id = e.id)))
     LEFT JOIN public.department_section ds ON ((ds.id = sc.section_id)))
     LEFT JOIN ( SELECT DISTINCT ON (employee_status.employee_id) employee_status.id,
            employee_status.employee_id,
            employee_status.status_id,
            employee_status.from_date
           FROM public.employee_status
          WHERE (employee_status.to_date IS NULL)
          ORDER BY employee_status.employee_id, employee_status.from_date DESC, employee_status.id DESC) s ON ((s.employee_id = e.id)))
     LEFT JOIN public.m_employee_status ms ON ((ms.id = s.status_id)));


ALTER TABLE public.view_employee_depart_status OWNER TO postgres;

-- Name: mat_view_employee_nik_period; Type: MATERIALIZED VIEW; Schema: public; Owner: postgres
--

CREATE MATERIALIZED VIEW public.mat_view_employee_nik_period AS
 SELECT e.nik,
    p.date_period,
    i."inout"
   FROM ( SELECT mat_view_employee_active.nik
           FROM public.mat_view_employee_active
          ORDER BY mat_view_employee_active.nik) e,
    ( SELECT (date_period.date_period)::date AS date_period
           FROM generate_series('2018-02-01 00:00:00'::timestamp without time zone, '2018-02-28 00:00:00'::timestamp without time zone, '1 day'::interval) date_period(date_period)) p,
    ( SELECT ("inout"."inout")::numeric AS "inout"
           FROM generate_series(0, 1) "inout"("inout")) i
  ORDER BY e.nik, p.date_period, i."inout"
  WITH NO DATA;

--
-- Name: v_material_budget_monitoring; Type: VIEW; Schema: public; Owner: postgres
--

CREATE VIEW public.v_material_budget_monitoring AS
 SELECT p.id AS project_id,
    p.name AS project_name,
    m.id AS material_id,
    m.material_code,
    m.material_name,
    m.unit,
    COALESCE(r.budget_qty, (0)::numeric) AS budget_qty,
    COALESCE(r.unit_price, (0)::numeric) AS unit_price,
    COALESCE((r.budget_qty * r.unit_price), (0)::numeric) AS total_budget_cost,
    COALESCE(mo.total_qty_out, (0)::numeric) AS total_qty_out,
    (COALESCE(mo.total_qty_out, (0)::numeric) * COALESCE(r.unit_price, (0)::numeric)) AS total_actual_cost,
    (COALESCE(r.budget_qty, (0)::numeric) - COALESCE(mo.total_qty_out, (0)::numeric)) AS remaining_qty,
        CASE
            WHEN (COALESCE(r.budget_qty, (0)::numeric) > (0)::numeric) THEN round(((COALESCE(mo.total_qty_out, (0)::numeric) / r.budget_qty) * (100)::numeric), 2)
            ELSE (0)::numeric
        END AS usage_pct,
        CASE
            WHEN (COALESCE(mo.total_qty_out, (0)::numeric) > (COALESCE(r.budget_qty, (0)::numeric) * ((1)::numeric + (COALESCE(r.tolerance_pct, (0)::numeric) / (100)::numeric)))) THEN 'OVERBUDGET'::text
            WHEN ((COALESCE(mo.total_qty_out, (0)::numeric) >= (COALESCE(r.budget_qty, (0)::numeric) * 0.85)) AND (COALESCE(r.budget_qty, (0)::numeric) > (0)::numeric)) THEN 'WARNING'::text
            ELSE 'ON TRACK'::text
        END AS status
   FROM (((public.projects p
     LEFT JOIN public.rab_materials r ON ((p.id = r.project_id)))
     LEFT JOIN public.materials m ON ((m.id = r.material_id)))
     LEFT JOIN ( SELECT h.project_id,
            d.material_id,
            sum(d.qty_out) AS total_qty_out
           FROM (public.material_out_items d
             JOIN public.material_out_header h ON ((h.id = d.material_out_id)))
          WHERE ((h.status)::text = 'posted'::text)
          GROUP BY h.project_id, d.material_id) mo ON (((mo.project_id = r.project_id) AND (mo.material_id = r.material_id))));


ALTER TABLE public.v_material_budget_monitoring OWNER TO postgres;

--
-- Name: view_grn_invoice_status; Type: VIEW; Schema: public; Owner: postgres
--

CREATE VIEW public.view_grn_invoice_status AS
 SELECT g.id AS grn_hdr_id,
    g.ref_no AS grn_ref_no,
    g.grn_date,
    g.po_id,
    g.supplier_id,
    COALESCE(gt.total_amount, (0)::numeric) AS grn_total,
    COALESCE(it.invoiced_amount, (0)::numeric) AS invoiced_amount,
    (COALESCE(gt.total_amount, (0)::numeric) - COALESCE(it.invoiced_amount, (0)::numeric)) AS outstanding_amount
   FROM ((public.grn_hdr g
     LEFT JOIN ( SELECT grn_dtl.grn_hdr_id,
            sum((grn_dtl.qty_received * grn_dtl.unit_cost)) AS total_amount
           FROM public.grn_dtl
          GROUP BY grn_dtl.grn_hdr_id) gt ON ((gt.grn_hdr_id = g.id)))
     LEFT JOIN ( SELECT pid.grn_hdr_id,
            sum(pid.matched_amount) AS invoiced_amount
           FROM (public.purchase_invoice_dtl pid
             JOIN public.purchase_invoice_hdr pih ON ((pih.id = pid.invoice_hdr_id)))
          WHERE ((pih.status)::text <> 'cancelled'::text)
          GROUP BY pid.grn_hdr_id) it ON ((it.grn_hdr_id = g.id)))
  WHERE ((g.status)::text = 'confirmed'::text);
